+def wav_info(raw):
+    def no(why):
+        return _bad(f"audio: only WAV audio is taken ({why}); PCM 8/16/24/32-bit or float 32/64-bit RIFF/WAVE")
+    if len(raw) < 12 or raw[:4] != b"RIFF" or raw[8:12] != b"WAVE":
+        raise no("this is not a RIFF/WAVE file")
+    fmt, o = None, 12
+    while o + 8 <= len(raw):
+        cid, cs = raw[o:o + 4], struct.unpack_from("<I", raw, o + 4)[0]
+        if cid == b"fmt " and o + 8 + 16 <= len(raw):
+            tag, ch, rate, _, _, bits = struct.unpack_from("<HHIIHH", raw, o + 8)
+            if tag == 0xFFFE and cs >= 26 and o + 8 + 26 <= len(raw):
+                tag = struct.unpack_from("<H", raw, o + 8 + 24)[0]
+            fmt = (tag, ch, rate, bits)
+        elif cid == b"data":
+            if fmt is None:
+                raise no("no fmt chunk before the data")
+            tag, ch, rate, bits = fmt
+            if not ((tag == 1 and bits in (8, 16, 24, 32)) or (tag == 3 and bits in (32, 64))):
+                raise no(f"format tag {tag}, {bits} bits")
+            if ch < 1 or rate < 1:
+                raise no("a zero channel count or sample rate")
+            n = len(raw) - (o + 8) if cs in (0, 0xFFFFFFFF) or cs > len(raw) - (o + 8) else cs
+            frames = n // (ch * (bits // 8))
+            if frames < 1:
+                raise _bad("audio: the WAV holds no samples")
+            return frames, rate, ch
+        o += 8 + cs + (cs & 1)
